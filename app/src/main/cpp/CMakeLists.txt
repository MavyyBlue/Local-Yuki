cmake_minimum_required(VERSION 3.22.1)
project(yuki_organs LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
set(GGML_CPU_ARM_ARCH "armv8-a" CACHE STRING "" FORCE)
add_subdirectory(vendor/llama.cpp)
if(ANDROID)
 add_library(yuki-organ SHARED organ.cpp)
 target_include_directories(yuki-organ PRIVATE vendor/llama.cpp/include)
 target_link_libraries(yuki-organ PRIVATE llama log)
endif()

# Host proof compiles the SAME JNI boundary; Android remains the shipping target.
if(NOT ANDROID AND YUKI_HOST_PROOF)
 find_package(JNI REQUIRED)
 add_library(yuki-organ SHARED organ.cpp)
 target_include_directories(yuki-organ PRIVATE ${JNI_INCLUDE_DIRS} vendor/llama.cpp/include)
 target_link_libraries(yuki-organ PRIVATE llama)
 if(YUKI_DESCRIPTOR_TESTS)
  add_library(yuki-descriptor-test SHARED ../../../../scripts/engine-proof/descriptor_sandbox.cpp)
  target_include_directories(yuki-descriptor-test PRIVATE ${JNI_INCLUDE_DIRS})
 endif()
endif()
